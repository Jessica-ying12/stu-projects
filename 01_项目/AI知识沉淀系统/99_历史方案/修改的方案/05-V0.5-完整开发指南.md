### Step 1：Embedding 服务

**文件：** `electron/services/embedding/index.ts`

```typescript
import OpenAI from 'openai';
import { database } from '../database';

export class EmbeddingService {
  private client: OpenAI;

  constructor(apiKey: string) {
    this.client = new OpenAI({ apiKey });
  }

  // 生成单个文本的 embedding
  async embed(text: string): Promise<number[]> {
    const response = await this.client.embeddings.create({
      model: 'text-embedding-3-small',
      input: text,
    });
    return response.data[0].embedding;
  }

  // 批量生成
  async embedBatch(texts: string[]): Promise<number[][]> {
    const response = await this.client.embeddings.create({
      model: 'text-embedding-3-small',
      input: texts,
    });
    return response.data.map(d => d.embedding);
  }

  // 为笔记生成并存储 embedding
  async processNote(noteId: string) {
    const note = await database.notes.get(noteId);
    if (!note) return;

    // 分块
    const chunks = this.splitText(note.content);

    // 删除旧的 embedding
    await database.embeddings.deleteByNote(noteId);

    // 生成新的 embedding
    for (let i = 0; i < chunks.length; i++) {
      const embedding = await this.embed(chunks[i]);
      await database.embeddings.create({
        noteId,
        chunkIndex: i,
        chunkText: chunks[i],
        embedding: Buffer.from(new Float32Array(embedding).buffer),
      });
    }
  }

  // 简单文本分块
  private splitText(text: string, chunkSize = 500, overlap = 50): string[] {
    const chunks: string[] = [];
    const sentences = text.split(/[。！？\n]+/);
    let currentChunk = '';

    for (const sentence of sentences) {
      if ((currentChunk + sentence).length > chunkSize && currentChunk) {
        chunks.push(currentChunk.trim());
        // overlap: 保留最后一部分
        const words = currentChunk.split('');
        currentChunk = words.slice(-overlap).join('') + sentence;
      } else {
        currentChunk += sentence + '。';
      }
    }

    if (currentChunk.trim()) {
      chunks.push(currentChunk.trim());
    }

    return chunks;
  }

  // 批量处理所有笔记
  async processAllNotes() {
    const notes = await database.notes.getAll();
    for (const note of notes) {
      await this.processNote(note.id);
    }
  }
}
```

### Step 2：向量检索

**文件：** `electron/services/retrieval/index.ts`

```typescript
import { cosine } from 'ml-distance/src/distances';
import { database } from '../database';
import { EmbeddingService } from '../embedding';

export interface RetrievalResult {
  noteId: string;
  noteTitle: string;
  chunkText: string;
  similarity: number;
  category?: string;
  tags?: string[];
}

export class RetrievalService {
  private embeddingService: EmbeddingService;

  constructor(embeddingService: EmbeddingService) {
    this.embeddingService = embeddingService;
  }

  async search(query: string, topK = 5, threshold = 0.3): Promise<RetrievalResult[]> {
    // 1. 生成查询的 embedding
    const queryEmbedding = await this.embeddingService.embed(query);

    // 2. 获取所有存储的 embeddings
    const allEmbeddings = await database.embeddings.getAll();

    // 3. 计算相似度
    const results: RetrievalResult[] = [];

    for (const item of allEmbeddings) {
      const storedEmbedding = new Float32Array(item.embedding.buffer);
      const similarity = 1 - cosine(queryEmbedding, Array.from(storedEmbedding));

      if (similarity >= threshold) {
        const note = await database.notes.get(item.noteId);
        results.push({
          noteId: item.noteId,
          noteTitle: note?.title || '',
          chunkText: item.chunkText,
          similarity,
          category: note?.category,
          tags: note?.tags,
        });
      }
    }

    // 4. 排序并返回 Top-K
    return results
      .sort((a, b) => b.similarity - a.similarity)
      .slice(0, topK);
  }
}
```

### Step 3：RAG 问答

**文件：** `electron/services/rag/index.ts`

```typescript
import { AIService } from '../ai';
import { RetrievalService } from '../retrieval';

export class RAGService {
  private ai: AIService;
  private retrieval: RetrievalService;

  constructor(ai: AIService, retrieval: RetrievalService) {
    this.ai = ai;
    this.retrieval = retrieval;
  }

  async answer(question: string, chatHistory: Array<{role: string; content: string}> = []) {
    // 1. 检索相关知识
    const results = await this.retrieval.search(question, 5, 0.3);

    // 2. 构建上下文
    const context = results.map((r, i) =>
      `[来源${i + 1}: ${r.noteTitle}]\n${r.chunkText}`
    ).join('\n\n---\n\n');

    // 3. 构建 prompt
    const systemPrompt = `你是用户的个人知识助手。请根据以下知识库内容回答用户的问题。

## 相关知识
${context || '（未找到相关知识）'}

## 规则
1. 只基于上述知识内容回答，不要编造信息
2. 如果知识库中没有相关内容，请明确告知"知识库中暂无相关内容"
3. 回答末尾用 [来源X] 标注引用
4. 保持简洁，直接回答问题`;

    // 4. 调用 AI
    const messages = [
      ...chatHistory.slice(-6), // 保留最近 3 轮对话
      { role: 'user', content: question },
    ];

    const answer = await this.ai.chat(
      messages.map(m => m.content).join('\n'),
      systemPrompt
    );

    return {
      answer,
      sources: results,
    };
  }

  // 流式回答
  async *answerStream(question: string, chatHistory: Array<{role: string; content: string}> = []) {
    const results = await this.retrieval.search(question, 5, 0.3);

    const context = results.map((r, i) =>
      `[来源${i + 1}: ${r.noteTitle}]\n${r.chunkText}`
    ).join('\n\n---\n\n');

    const systemPrompt = `你是用户的个人知识助手...`; // 同上

    // 流式输出
    yield { type: 'sources', data: results };

    for await (const chunk of this.ai.chatStream(question, systemPrompt)) {
      yield { type: 'text', data: chunk };
    }
  }
}
```

### Step 4：问答界面

**文件：** `src/pages/chat/ChatPage.tsx`

```typescript
// 核心组件结构
function ChatPage() {
  const [messages, setMessages] = useState<ChatMessage[]>([]);
  const [input, setInput] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const [sources, setSources] = useState<RetrievalResult[]>([]);

  async function handleSend() {
    if (!input.trim() || isLoading) return;

    const userMessage = { role: 'user', content: input };
    setMessages(prev => [...prev, userMessage]);
    setInput('');
    setIsLoading(true);

    try {
      const result = await window.api.invoke('rag:answer', {
        question: input,
        history: messages.slice(-6),
      });

      setMessages(prev => [...prev, {
        role: 'assistant',
        content: result.answer,
        sources: result.sources,
      }]);
      setSources(result.sources);
    } finally {
      setIsLoading(false);
    }
  }

  return (
    <div className="flex h-full">
      {/* 左侧：聊天区域 */}
      <div className="flex-1 flex flex-col">
        <ChatMessages messages={messages} />
        <ChatInput value={input} onChange={setInput} onSend={handleSend} loading={isLoading} />
      </div>
      {/* 右侧：引用来源面板 */}
      <SourcePanel sources={sources} />
    </div>
  );
}
```

### Step 5：相关笔记推荐

```typescript
// src/components/RelatedNotes.tsx
function RelatedNotes({ noteId }) {
  const [related, setRelated] = useState([]);

  useEffect(() => {
    window.api.invoke('retrieval:find-similar', { noteId, topK: 5 })
      .then(setRelated);
  }, [noteId]);

  if (!related.length) return null;

  return (
    <div className="mt-8 border-t pt-4">
      <h3 className="font-medium mb-3">相关笔记</h3>
      <div className="grid grid-cols-1 gap-2">
        {related.map(note => (
          <Link key={note.id} to={`/notes/${note.id}`}
                className="p-3 bg-gray-50 rounded hover:bg-gray-100">
            <div className="font-medium text-sm">{note.title}</div>
            <div className="text-xs text-gray-500 mt-1">{note.category}</div>
          </Link>
        ))}
      </div>
    </div>
  );
}
```

```typescript
function DailyReview() {
  const [notes, setNotes] = useState([]);
```
