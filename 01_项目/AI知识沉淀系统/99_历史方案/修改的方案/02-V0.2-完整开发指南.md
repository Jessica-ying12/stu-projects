## 详细实现指南

### Step 1：通用采集框架

**文件：** `electron/services/collector/base.ts`

```typescript
import { EventEmitter } from 'events';

export interface CollectorConfig {
  name: string;
  source: string;
  enabled: boolean;
  paths: string[];
  filePatterns: string[];
}

export interface ParsedConversation {
  source: string;
  title?: string;
  messages: Array<{
    role: 'user' | 'assistant' | 'system';
    content: string;
    timestamp?: string;
  }>;
  metadata: {
    filePath: string;
    workspace?: string;
    project?: string;
    toolVersion?: string;
  };
  createdAt: string;
  updatedAt: string;
}

export abstract class BaseCollector extends EventEmitter {
  abstract name: string;
  abstract source: string;
  protected config: CollectorConfig;

  constructor(config: CollectorConfig) {
    super();
    this.config = config;
  }

  abstract discoverFiles(): Promise<string[]>;
  abstract parseFile(filePath: string): Promise<ParsedConversation | null>;
  abstract getDefaultPaths(): string[];

  // 子类可覆盖
  isRelevantFile(filePath: string): boolean {
    return this.config.filePatterns.some(pattern =>
      filePath.includes(pattern) || filePath.endsWith(pattern)
    );
  }
}
```

**文件：** `electron/services/collector/registry.ts`

```typescript
import { BaseCollector } from './base';

class CollectorRegistry {
  private collectors: Map<string, BaseCollector> = new Map();

  register(collector: BaseCollector) {
    this.collectors.set(collector.source, collector);
  }

  get(source: string): BaseCollector | undefined {
    return this.collectors.get(source);
  }

  getAll(): BaseCollector[] {
    return Array.from(this.collectors.values());
  }

  getEnabled(): BaseCollector[] {
    return this.getAll().filter(c => c['config'].enabled);
  }
}

export const registry = new CollectorRegistry();
```

```typescript
import * as fs from 'fs';

export interface DetectedTool {

  source: string;
  detected: boolean;
  paths: string[];
  version?: string;

export function detectInstalledTools(): DetectedTool[] {
  const home = os.homedir();
  const appSupport = path.join(home, 'Library/Application Support');

  const tools: DetectedTool[] = [
    {
      name: 'Claude Code',
      source: 'claude-code',
      detected: fs.existsSync(path.join(home, '.claude')),
      paths: [path.join(home, '.claude/projects')],
    },
    {
      name: 'Codex',
      source: 'codex',
      detected: fs.existsSync(path.join(home, '.codex')),
      paths: [path.join(home, '.codex')],
    },
    {
      name: 'Cursor',
      source: 'cursor',
      detected: fs.existsSync(path.join(appSupport, 'Cursor')),
      paths: [path.join(appSupport, 'Cursor')],
    },
    {
      name: 'Kiro',
      source: 'kiro',
      detected: fs.existsSync(path.join(home, '.kiro'))
        || fs.existsSync(path.join(appSupport, 'Kiro')),
      paths: [
        path.join(home, '.kiro'),
        path.join(appSupport, 'Kiro'),
      ].filter(p => fs.existsSync(p)),
    },
    {
      name: 'Trea',
      source: 'trea',
      detected: fs.existsSync(path.join(appSupport, 'Trea')),
      paths: [path.join(appSupport, 'Trea')],
    },
  ];

  return tools;
```

### Step 4：文件监控服务

**文件：** `electron/services/collector/watcher.ts`

```typescript
import chokidar from 'chokidar';
import { BaseCollector } from './base';
import { registry } from './registry';

export class FileWatcher {
  private watchers: Map<string, chokidar.FSWatcher> = new Map();

  startWatching(collector: BaseCollector, paths: string[]) {
    const watcher = chokidar.watch(paths, {
      persistent: true,
      ignoreInitial: true,  // 不触发已有文件
      awaitWriteFinish: {   // 等待文件写入完成
        stabilityThreshold: 2000,
        pollInterval: 100,
      },
    });

    watcher.on('add', (filePath) => this.handleNewFile(collector, filePath));
    watcher.on('change', (filePath) => this.handleFileChange(collector, filePath));
    watcher.on('error', (error) => console.error(`Watcher error: ${error}`));

    this.watchers.set(collector.source, watcher);
  }

  private async handleNewFile(collector: BaseCollector, filePath: string) {
    if (!collector.isRelevantFile(filePath)) return;

    const conversation = await collector.parseFile(filePath);
    if (conversation) {
      collector.emit('new-conversation', conversation);
    }
  }

  private async handleFileChange(collector: BaseCollector, filePath: string) {
    if (!collector.isRelevantFile(filePath)) return;

    const conversation = await collector.parseFile(filePath);
    if (conversation) {
      collector.emit('updated-conversation', conversation);
    }
  }

  stopWatching(source: string) {
    const watcher = this.watchers.get(source);
    if (watcher) {
      watcher.close();
      this.watchers.delete(source);
    }
  }

  stopAll() {
    for (const [source, watcher] of this.watchers) {
      watcher.close();
    }
    this.watchers.clear();
  }
}
```

### Step 8：采集管理器

**文件：** `electron/services/collector/manager.ts`

```typescript
import { registry } from './registry';
import { FileWatcher } from './watcher';
import { database } from '../database';

export class CollectorManager {
  private watcher: FileWatcher;
  private status: 'running' | 'paused' | 'stopped' = 'stopped';

  constructor() {
    this.watcher = new FileWatcher();
  }

  async start() {
    const collectors = registry.getEnabled();

    for (const collector of collectors) {
      const paths = collector.getDefaultPaths();

      // 监听新对话事件
      collector.on('new-conversation', async (conv) => {
        await database.conversations.create(conv);
        this.notify(conv);
      });

      collector.on('updated-conversation', async (conv) => {
        await database.conversations.upsert(conv);
      });

      // 启动监控
      this.watcher.startWatching(collector, paths);
    }

    this.status = 'running';
  }

  pause() {
    this.watcher.stopAll();
    this.status = 'paused';
  }

  resume() {
    this.start();
  }

  stop() {
    this.watcher.stopAll();
    this.status = 'stopped';
  }

  getStatus() {
    return this.status;
  }

  // 手动全量扫描
  async fullScan() {
    const collectors = registry.getEnabled();
    for (const collector of collectors) {
      const files = await collector.discoverFiles();
      for (const file of files) {
        const conv = await collector.parseFile(file);
        if (conv) {
          await database.conversations.upsert(conv);
        }
      }
    }
  }

  private notify(conv: any) {
    // 通过 Electron Notification API 通知
  }
}
```

## 注意事项

1. **存储格式可能变化** - AI 工具更新频繁，存储格式可能随版本变化。每个解析器需要做好版本兼容和错误容忍。

2. **文件锁** - 某些工具可能在写入时锁定文件，读取时需要处理 EBUSY 错误。

3. **性能** - 不要一次性读取大文件到内存，使用流式读取。

4. **隐私** - 对话可能包含敏感信息，采集后存储在本地，不要上传。

5. **首次全量扫描** - 用户可能已有大量历史对话，首次扫描需要做好进度提示和后台处理。
