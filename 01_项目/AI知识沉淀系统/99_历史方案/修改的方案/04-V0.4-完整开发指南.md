### Step 5：知识去重

**简单相似度算法（Jaccard 系数）：**

```typescript
// electron/services/processor/deduplicator.ts

function extractKeywords(text: string): Set<string> {
  // 简单分词：按空格和标点分割，去停用词
  const words = text
    .replace(/[^\u4e00-\u9fa5a-zA-Z0-9\s]/g, ' ')
    .split(/\s+/)
    .filter(w => w.length > 1);
  return new Set(words);
}

function jaccardSimilarity(setA: Set<string>, setB: Set<string>): number {
  const intersection = new Set([...setA].filter(x => setB.has(x)));
  const union = new Set([...setA, ...setB]);
  return intersection.size / union.size;
}

export async function findSimilarNotes(newContent: string, threshold = 0.7) {
  const newKeywords = extractKeywords(newContent);
  const existingNotes = await database.notes.getAll();

  const similar = [];
  for (const note of existingNotes) {
    const noteKeywords = extractKeywords(note.content);
    const similarity = jaccardSimilarity(newKeywords, noteKeywords);
    if (similarity >= threshold) {
      similar.push({ note, similarity });
    }
  }

  return similar.sort((a, b) => b.similarity - a.similarity);
}
```

### Step 6：导出功能

**Markdown 导出：**

```typescript
// electron/services/exporter/markdown.ts
import { dialog } from 'electron';
import * as fs from 'fs/promises';
import * as path from 'path';

export async function exportToMarkdown(note: Note) {
  const { filePath } = await dialog.showSaveDialog({
    defaultPath: `${note.title}.md`,
    filters: [{ name: 'Markdown', extensions: ['md'] }],
  });

  if (!filePath) return;

  const frontmatter = `---
title: ${note.title}
category: ${note.category}
tags: [${note.tags.join(', ')}]
created: ${note.createdAt}
---

`;

  await fs.writeFile(filePath, frontmatter + note.content, 'utf-8');
}

export async function exportBatchToMarkdown(notes: Note[]) {
  const { filePaths } = await dialog.showOpenDialog({
    properties: ['openDirectory', 'createDirectory'],
    title: '选择导出目录',
  });

  if (!filePaths?.[0]) return;
  const outputDir = filePaths[0];

  for (const note of notes) {
    const category = note.category || '未分类';
    const dir = path.join(outputDir, category);
    await fs.mkdir(dir, { recursive: true });

    const fileName = `${note.title.replace(/[/\\?%*:|"<>]/g, '-')}.md`;
    await fs.writeFile(path.join(dir, fileName), note.content, 'utf-8');
  }
}
```

// electron/services/exporter/pdf.ts
import { BrowserWindow } from 'electron';

export async function exportToPDF(note: Note) {
  // 1. 先生成 HTML
  const html = await generateHTML(note);

  // 2. 创建隐藏窗口加载 HTML
  const win = new BrowserWindow({ show: false, width: 800, height: 600 });
  await win.loadURL(`data:text/html;charset=utf-8,${encodeURIComponent(html)}`);

  // 3. 导出 PDF
  const pdfBuffer = await win.webContents.printToPDF({
    pageSize: 'A4',
    printBackground: true,
    margins: { top: 20, bottom: 20, left: 20, right: 20 },
  });

  // 4. 保存文件
  const { filePath } = await dialog.showSaveDialog({
    defaultPath: `${note.title}.pdf`,
    filters: [{ name: 'PDF', extensions: ['pdf'] }],
  });

  if (filePath) {
    await fs.writeFile(filePath, pdfBuffer);
  }

  // 5. 关闭隐藏窗口
  win.close();
}
