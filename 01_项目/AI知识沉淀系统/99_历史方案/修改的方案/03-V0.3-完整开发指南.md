### Step 1：大模型 API 接入层

**文件：** `electron/services/ai/index.ts`

```typescript
import { ClaudeProvider } from './claude';
import { OpenAIProvider } from './openai';

export interface AIProvider {
  name: string;
  chat(messages: Array<{role: string; content: string}>): Promise<string>;
  chatStream(messages: Array<{role: string; content: string}>): AsyncIterable<string>;
}

export class AIService {
  private provider: AIProvider;

  constructor(providerName: string, apiKey: string) {
    switch (providerName) {
      case 'claude':
        this.provider = new ClaudeProvider(apiKey);
        break;
      case 'openai':
        this.provider = new OpenAIProvider(apiKey);
        break;
      default:
        throw new Error(`Unknown provider: ${providerName}`);
    }
  }

  async chat(prompt: string, systemPrompt?: string): Promise<string> {
    const messages = [];
    if (systemPrompt) {
      messages.push({ role: 'system', content: systemPrompt });
    }
    messages.push({ role: 'user', content: prompt });
    return this.provider.chat(messages);
  }
}
```

**文件：** `electron/services/ai/claude.ts`

```typescript
import Anthropic from '@anthropic-ai/sdk';
import { AIProvider } from './index';

export class ClaudeProvider implements AIProvider {
  name = 'claude';
  private client: Anthropic;

  constructor(apiKey: string) {
    this.client = new Anthropic({ apiKey });
  }

  async chat(messages: Array<{role: string; content: string}>): Promise<string> {
    const systemMsg = messages.find(m => m.role === 'system');
    const userMsgs = messages.filter(m => m.role !== 'system');

    const response = await this.client.messages.create({
      model: 'claude-sonnet-4-20250514',
      max_tokens: 2048,
      system: systemMsg?.content,
      messages: userMsgs.map(m => ({
        role: m.role as 'user' | 'assistant',
        content: m.content,
      })),
    });

    return response.content[0].type === 'text' ? response.content[0].text : '';
  }

  async *chatStream(messages: Array<{role: string; content: string}>): AsyncIterable<string> {
    const systemMsg = messages.find(m => m.role === 'system');
    const userMsgs = messages.filter(m => m.role !== 'system');

    const stream = this.client.messages.stream({
      model: 'claude-sonnet-4-20250514',
      max_tokens: 2048,
      system: systemMsg?.content,
      messages: userMsgs.map(m => ({
        role: m.role as 'user' | 'assistant',
        content: m.content,
      })),
    });

    for await (const event of stream) {
      if (event.type === 'content_block_delta' && event.delta.type === 'text_delta') {
        yield event.delta.text;
      }
    }
  }
}
```

**文件：** `electron/services/ai/openai.ts`

```typescript
import OpenAI from 'openai';
import { AIProvider } from './index';

export class OpenAIProvider implements AIProvider {
  name = 'openai';
  private client: OpenAI;

  constructor(apiKey: string) {
    this.client = new OpenAI({ apiKey });
  }

  async chat(messages: Array<{role: string; content: string}>): Promise<string> {
    const response = await this.client.chat.completions.create({
      model: 'gpt-4o-mini',  // 整理任务用小模型即可
      messages: messages as any,
    });
    return response.choices[0]?.message?.content || '';
  }

  async *chatStream(messages: Array<{role: string; content: string}>): AsyncIterable<string> {
    const stream = await this.client.chat.completions.create({
      model: 'gpt-4o-mini',
      messages: messages as any,
      stream: true,
    });

    for await (const chunk of stream) {
      const content = chunk.choices[0]?.delta?.content;
      if (content) yield content;
    }
  }
}
```

### Step 2：处理队列

**文件：** `electron/services/processor/queue.ts`

```typescript
import PQueue from 'p-queue';
import { database } from '../database';

export type TaskType = 'title' | 'summary' | 'category' | 'tags' | 'note';

interface ProcessingTask {
  id: string;
  conversationId: string;
  taskType: TaskType;
  status: 'pending' | 'processing' | 'completed' | 'failed';
  retryCount: number;
}

export class ProcessingQueue {
  private queue: PQueue;

  constructor(concurrency = 2) {
    this.queue = new PQueue({ concurrency });
  }

  async addTask(conversationId: string, taskType: TaskType) {
    const task = await database.processingTasks.create({
      conversationId,
      taskType,
      status: 'pending',
      retryCount: 0,
    });

    this.queue.add(async () => {
      await this.processTask(task);
    });
  }

  // 新对话进来时，添加所有处理任务
  async addAllTasks(conversationId: string) {
    const tasks: TaskType[] = ['title', 'summary', 'category', 'tags', 'note'];
    // title 和 summary 先做，其他依赖分类结果
    await this.addTask(conversationId, 'title');
    await this.addTask(conversationId, 'summary');
    await this.addTask(conversationId, 'category');
    await this.addTask(conversationId, 'tags');
    await this.addTask(conversationId, 'note');
  }

  private async processTask(task: ProcessingTask) {
    try {
      await database.processingTasks.updateStatus(task.id, 'processing');

      // 根据任务类型调用不同处理器
      switch (task.taskType) {
        case 'title':
          await this.processTitle(task.conversationId);
          break;
        case 'summary':
          await this.processSummary(task.conversationId);
          break;
        case 'category':
          await this.processCategory(task.conversationId);
          break;
        case 'tags':
          await this.processTags(task.conversationId);
          break;
        case 'note':
          await this.processNote(task.conversationId);
          break;
      }

      await database.processingTasks.updateStatus(task.id, 'completed');
    } catch (error) {
      if (task.retryCount < 3) {
        await database.processingTasks.retry(task.id);
        this.queue.add(() => this.processTask({ ...task, retryCount: task.retryCount + 1 }));
      } else {
        await database.processingTasks.updateStatus(task.id, 'failed', error.message);
      }
    }
  }

  private async processTitle(conversationId: string) { /* 调用 AI */ }
  private async processSummary(conversationId: string) { /* 调用 AI */ }
  private async processCategory(conversationId: string) { /* 调用 AI */ }
  private async processTags(conversationId: string) { /* 调用 AI */ }
  private async processNote(conversationId: string) { /* 调用 AI */ }

  getStats() {
    return {
      pending: this.queue.pending,
      size: this.queue.size,
    };
  }
}
```

## API Key 安全存储

使用 Electron 的 `safeStorage` API：

```typescript
import { safeStorage } from 'electron';

export function encryptApiKey(key: string): string {
  const buffer = safeStorage.encryptString(key);
  return buffer.toString('base64');
}

export function decryptApiKey(encrypted: string): string {
  const buffer = Buffer.from(encrypted, 'base64');
  return safeStorage.decryptString(buffer);
}
```
